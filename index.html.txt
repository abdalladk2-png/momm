    <script>
        // دالة إطلاق القلوب والاحتفال
        function burstHearts() {
            confetti({
                particleCount: 80,
                spread: 70,
                origin: { y: 0.6 },
                colors: ['#ff0000', '#ff69b4', '#ff1493', '#ffffff']
            });
            
            var song = document.getElementById('momSong');
            if (song.paused) {
                song.play();
            }
        }

        // إنشاء قلوب متساقطة في الخلفية
        function createFloatingHeart() {
            const container = document.getElementById('hearts-container');
            const heart = document.createElement('i');
            heart.classList.add('floating-heart', 'fas', 'fa-heart');
            
            heart.style.left = Math.random() * 100 + 'vw';
            heart.style.fontSize = (Math.random() * 20 + 15) + 'px';
            heart.style.animationDuration = (Math.random() * 3 + 3) + 's';
            
            container.appendChild(heart);

            setTimeout(() => {
                heart.remove();
            }, 5000);
        }

        // توليد قلب كل 300 مللي ثانية
        setInterval(createFloatingHeart, 300);

        // إطلاق احتفال خفيف أول ما الصفحة تفتح
        window.onload = function() {
            setTimeout(burstHearts, 500);
        };
    </script>
